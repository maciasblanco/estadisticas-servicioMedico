Public Function TieneRol(ByVal rol As String) As Boolean
    TieneRol = (UCase$(gRolUsuario) = UCase$(rol))
End Function
Public Function EsAdmin() As Boolean
    EsAdmin = TieneRol(ROL_ADMIN)
End Function
Public Function TienePermiso(ByVal accion As String) As Boolean
    Select Case UCase$(accion)
        Case "INSERT", "UPDATE", "DELETE"
            TienePermiso = EsAdmin() Or TieneRol(ROL_MEDICO) Or TieneRol(ROL_ESTADISTICA)
        Case "EXPORT"
            TienePermiso = EsAdmin() Or TieneRol(ROL_ESTADISTICA)
        Case "USUARIOS", "RESPALDO", "CONFIG"
            TienePermiso = EsAdmin()
        Case Else
            TienePermiso = True
    End Select
End Function

Public Sub RangoMes(ByVal anio As Integer, ByVal mes As Integer, ByRef fIni As Date, ByRef fFin As Date)

    fFin = DateSerial(anio, mes + 1, 0)
End Sub
Public Sub RangoTrimestre(ByVal anio As Integer, ByVal trimestre As Integer, ByRef fIni As Date, ByRef fFin As Date)
    Dim mesIni As Integer
    mesIni = ((trimestre - 1) * 3) + 1
    fIni = DateSerial(anio, mesIni, 1)
    fFin = DateSerial(anio, mesIni + 3, 0)
End Sub
Public Sub RangoSemestre(ByVal anio As Integer, ByVal semestre As Integer, ByRef fIni As Date, ByRef fFin As Date)
    If semestre = 1 Then
        fIni = DateSerial(anio, 1, 1)
        fFin = DateSerial(anio, 6, 30)
    Else
        fIni = DateSerial(anio, 7, 1)
        fFin = DateSerial(anio, 12, 31)
    End If
End Sub

    Dim a As Long, b As Long, c As Long, d As Long

    For j = 7 To 0 Step -1

    Next

Private Function RShift(ByVal v As Long, ByVal n As Integer) As Long
    Dim d As Double
    d = v
    If d < 0 Then d = d + 4294967296#
    RShift = Int(d / (2 ^ n)) And &HFFFFFFFF
    If RShift < 0 Then RShift = RShift + 4294967296#
End Function
Public Function ValidarLogin(ByVal usuario As String, ByVal clave As String, _
                              ByRef mensaje As String) As Boolean
    ValidarLogin = False
    If Not ExisteHoja(HOJA_USUARIOS) Then
        mensaje = "No existe la hoja de usuarios."
        Exit Function
    End If
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim usuarioLimpio As String
    Dim intentos As Integer
    Dim hashGuardado As String
    Set ws = ThisWorkbook.Sheets(HOJA_USUARIOS)
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    usuarioLimpio = UCase(Trim(usuario))
    For i = 2 To lastRow
        If UCase(Trim(CStr(ws.Cells(i, UCOL_USUARIO).Value))) = usuarioLimpio Then
            If UCase(Trim(CStr(ws.Cells(i, UCOL_ACTIVO).Value))) <> "SI" Then
                mensaje = "Usuario deshabilitado. Contacte al administrador."
                Exit Function
            End If
            intentos = Val(CStr(ws.Cells(i, UCOL_INTENTOS).Value))
            If intentos >= MAX_INTENTOS Then
                mensaje = "Usuario bloqueado por " & MAX_INTENTOS & " intentos fallidos."
                Exit Function
            End If
            hashGuardado = CStr(ws.Cells(i, UCOL_HASH).Value)
            If hashGuardado = "INICIAL" Then
                If clave = "Admin2026!" Then
                    ws.Cells(i, UCOL_HASH).Value = GenerarHash(clave)
                    ws.Cells(i, UCOL_INTENTOS).Value = 0
                    ws.Cells(i, UCOL_ULTIMO_ACCESO).Value = Format(Now, FORMATO_DATETIME)
                    InicializarSesion usuarioLimpio, CStr(ws.Cells(i, UCOL_NOMBRE).Value), CStr(ws.Cells(i, UCOL_ROL).Value)
                    mensaje = "OK"
                    ValidarLogin = True
                    Exit Function
                End If
            ElseIf LCase(GenerarHash(clave)) = LCase(hashGuardado) Then
                ws.Cells(i, UCOL_INTENTOS).Value = 0
                ws.Cells(i, UCOL_ULTIMO_ACCESO).Value = Format(Now, FORMATO_DATETIME)
                InicializarSesion usuarioLimpio, CStr(ws.Cells(i, UCOL_NOMBRE).Value), CStr(ws.Cells(i, UCOL_ROL).Value)
                mensaje = "OK"
                ValidarLogin = True
                Exit Function
            End If
            ws.Cells(i, UCOL_INTENTOS).Value = intentos + 1
            mensaje = "Clave incorrecta. Intentos: " & (intentos + 1) & " de " & MAX_INTENTOS
            Exit Function
        End If
    Next i
    mensaje = "Usuario no encontrado."
End Function
Public Function CambiarClave(ByVal usuario As String, ByVal claveNueva As String) As Boolean
    If Len(claveNueva) < MIN_LONG_CLAVE Then Exit Function
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Set ws = ThisWorkbook.Sheets(HOJA_USUARIOS)
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    For i = 2 To lastRow
        If UCase(Trim(CStr(ws.Cells(i, UCOL_USUARIO).Value))) = UCase(Trim(usuario)) Then
            ws.Cells(i, UCOL_HASH).Value = GenerarHash(claveNueva)
            ws.Cells(i, UCOL_INTENTOS).Value = 0
            CambiarClave = True
            Exit Function
        End If
    Next
End Function

Public Sub RegistrarHistorico(ByVal accion As String, ByVal hoja As String, ByVal idRegistro As Variant, ByVal campo As String, ByVal valorAnt As Variant, ByVal valorNuevo As Variant, Optional ByVal detalle As String = "")

Option Explicit
Public Function BuscarRegistros(ByVal criterio As String, ByRef resultados() As Variant) As Long
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets(HOJA_DATOS)
    Dim lastRow As Long: lastRow = ws.Cells(ws.Rows.Count, COL_CEDULA).End(xlUp).Row
    Dim count As Long: count = 0
    criterio = UCase(Trim(criterio))
    Dim i As Long
    For i = 2 To lastRow
        Dim textoFila As String
        textoFila = UCase(CStr(ws.Cells(i, COL_NOMBRE).Value) & "|" & CStr(ws.Cells(i, COL_CEDULA).Value))
        If Len(criterio) = 0 Or InStr(1, textoFila, criterio, vbTextCompare) > 0 Then
            count = count + 1
        End If
    Next i
    If count = 0 Then
        BuscarRegistros = 0
        Exit Function
    End If
    ReDim resultados(1 To count, 1 To 10)
    Dim k As Long: k = 0
    For i = 2 To lastRow
        Dim textoFila2 As String
        textoFila2 = UCase(CStr(ws.Cells(i, COL_NOMBRE).Value) & "|" & CStr(ws.Cells(i, COL_CEDULA).Value))
        If Len(criterio) = 0 Or InStr(1, textoFila2, criterio, vbTextCompare) > 0 Then
            k = k + 1
            resultados(k, 1) = ws.Cells(i, COL_FECHA).Value
            resultados(k, 2) = ws.Cells(i, COL_NOMBRE).Value
            resultados(k, 3) = ws.Cells(i, COL_CEDULA).Value
            resultados(k, 4) = ws.Cells(i, COL_SEXO).Value
            resultados(k, 5) = ws.Cells(i, COL_EDAD).Value
            resultados(k, 6) = ws.Cells(i, COL_ATENCION).Value
            resultados(k, 7) = ws.Cells(i, COL_MEDICO).Value
            resultados(k, 8) = ws.Cells(i, COL_CANT).Value
            resultados(k, 9) = ws.Cells(i, COL_PROCEDENCIA).Value
            resultados(k, 10) = ws.Cells(i, COL_DIAGNOSTICO).Value
        End If
    Next i
    BuscarRegistros = count
End Function
Public Function InsertarRegistro(ByRef valores() As Variant) As Boolean
    On Error GoTo Errores
    If Not TienePermiso("INSERT") Then
        MsgBox MSG_SIN_PERMISO, vbExclamation, "Permiso Denegado"
        InsertarRegistro = False: Exit Function
    End If
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets(HOJA_DATOS)
    Dim fila As Long: fila = ws.Cells(ws.Rows.Count, COL_CEDULA).End(xlUp).Row + 1
    Dim i As Long
    For i = LBound(valores) To UBound(valores)
        ws.Cells(fila, i - LBound(valores) + 1).Value = valores(i)
    Next i
    RegistrarHistorico ACC_INSERT, HOJA_DATOS, fila, "REGISTRO COMPLETO", "", Join(valores, " | "), "Registro creado"
    InsertarRegistro = True
    Exit Function
Errores:
    MsgBox "Error al insertar: " & Err.Description, vbCritical: InsertarRegistro = False
End Function
Public Function ActualizarRegistro(ByVal fila As Long, ByRef valores() As Variant) As Boolean
    On Error GoTo Errores
    If Not TienePermiso("UPDATE") Then
        MsgBox MSG_SIN_PERMISO, vbExclamation, "Permiso Denegado"
        ActualizarRegistro = False: Exit Function
    End If
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets(HOJA_DATOS)
    Dim i As Long, col As Long
    For i = LBound(valores) To UBound(valores)
        col = i - LBound(valores) + 1
        Dim valAnt As Variant: valAnt = ws.Cells(fila, col).Value
        If CStr(valAnt) <> CStr(valores(i)) Then
            ws.Cells(fila, col).Value = valores(i)
            RegistrarHistorico ACC_UPDATE, HOJA_DATOS, fila, CStr(ws.Cells(FILA_ENCABEZADO, col).Value), valAnt, valores(i), "Modificación"
        End If
    Next i
    ActualizarRegistro = True
    Exit Function
Errores:
    MsgBox "Error al actualizar: " & Err.Description, vbCritical: ActualizarRegistro = False
End Function
Public Function EliminarRegistro(ByVal fila As Long) As Boolean
    On Error GoTo Errores
    If Not TienePermiso("DELETE") Then
        MsgBox MSG_SIN_PERMISO, vbExclamation, "Permiso Denegado"
        EliminarRegistro = False: Exit Function
    End If
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets(HOJA_DATOS)
    Dim backup As String
    Dim i As Long
    For i = 1 To 10
        backup = backup & CStr(ws.Cells(fila, i).Value) & " | "
    Next i
    RegistrarHistorico ACC_DELETE, HOJA_DATOS, fila, "REGISTRO COMPLETO", backup, "", "Registro eliminado"
    ws.Rows(fila).Delete Shift:=xlUp
    EliminarRegistro = True
    Exit Function
Errores:
    MsgBox "Error al eliminar: " & Err.Description, vbCritical: EliminarRegistro = False
End Function

Option Explicit
Public Sub CalcularEstadisticas(ByVal fIni As Date, ByVal fFin As Date, ByRef txtResultado As String, ByRef totalReg As Long)
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets(HOJA_DATOS)
    Dim lastRow As Long: lastRow = ws.Cells(ws.Rows.Count, COL_FECHA).End(xlUp).Row
    Dim i As Long
    Dim xSexo As Object, xAtencion As Object, xSistema As Object, xMedico As Object
    Set xSexo = CreateObject("Scripting.Dictionary")
    Set xAtencion = CreateObject("Scripting.Dictionary")
    Set xSistema = CreateObject("Scripting.Dictionary")
    Set xMedico = CreateObject("Scripting.Dictionary")
    For i = 2 To lastRow
        Dim f As Variant: f = ws.Cells(i, COL_FECHA).Value
        If IsDate(f) Then
            If CDate(f) >= fIni And CDate(f) <= fFin Then
                totalReg = totalReg + 1
                SumarDic xSexo, CStr(ws.Cells(i, COL_SEXO).Value)
                SumarDic xAtencion, UCase(CStr(ws.Cells(i, COL_ATENCION).Value))
                SumarDic xSistema, UCase(CStr(ws.Cells(i, COL_DIAGNOSTICO).Value))
                SumarDic xMedico, UCase(CStr(ws.Cells(i, COL_MEDICO).Value))
            End If
        End If
    Next i
    txtResultado = "TOTAL DE CONSULTAS: " & totalReg & vbCrLf & vbCrLf
    txtResultado = txtResultado & "▶ POR SEXO" & vbCrLf & Desglosar(xSexo) & vbCrLf
    txtResultado = txtResultado & "▶ POR TIPO DE ATENCIÓN" & vbCrLf & Desglosar(xAtencion) & vbCrLf
    txtResultado = txtResultado & "▶ POR SISTEMA AFECTADO" & vbCrLf & Desglosar(xSistema) & vbCrLf
    txtResultado = txtResultado & "▶ POR MÉDICO TRATANTE" & vbCrLf & Desglosar(xMedico)
End Sub
Private Sub SumarDic(d As Object, clave As String)
    If Len(clave) = 0 Then clave = "(SIN DATO)"
    If d.Exists(clave) Then
        d(clave) = d(clave) + 1
    Else
        d.Add clave, 1
    End If
End Sub
Private Function Desglosar(d As Object) As String
    Dim k As Variant, s As String
    For Each k In d.Keys
        s = s & "   • " & k & ": " & d(k) & vbCrLf
    Next k
    Desglosar = s
End Function

        Case "Por Trimestre"

            RangoSemestre anio, semestre, fIni, fFin
    End Select
    Dim reporte As String, total As Long
    CalcularEstadisticas fIni, fFin, reporte, total
